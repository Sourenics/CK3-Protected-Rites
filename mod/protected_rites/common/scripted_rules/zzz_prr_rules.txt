# Root is the rite editor

	can_edit_rite_trigger = yes
	# PROTECTED RITES: an AI Head of Rite cannot edit a protected rite. Players keep the vanilla rules.
	trigger_if = {
		limit = { is_ai = yes }
		rite = { prr_rite_is_protected_trigger = no }
	}
}
