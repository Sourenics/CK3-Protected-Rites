# Protected Rites - scripted triggers.

# Rite scope. A rite is protected while a player follows it, as long as it is not one of the game's
# own rites: either a player (or a player's theological puppet) founded it, or it has no rite type.
# With the game rule set to vanilla, a protected rite loses its protection while it is the mainline rite.
prr_rite_is_protected_trigger = {
	OR = {
		has_variable = prr_player_founded_rite
		NOT = { exists = rite_type }
	}
	trigger_if = {
		limit = { has_game_rule = prr_mainline_vanilla }
		this != faith.main_rite
	}
	save_temporary_scope_as = prr_checked_rite
	any_player = {
		rite = scope:prr_checked_rite
	}
}

# Character scope. A player, or the theological puppet of a player.
prr_is_player_or_player_puppet_trigger = {
	OR = {
		is_ai = no
		puppeteer ?= { is_ai = no }
	}
}

# Tenet scope. The tenet is a core tenet of a protected rite of $FAITH$.
prr_tenet_is_core_in_protected_rite_trigger = {
	save_temporary_scope_as = prr_checked_tenet
	$FAITH$ = {
		any_faith_rite = {
			rite_has_tenet = scope:prr_checked_tenet
			prr_rite_is_protected_trigger = yes
		}
	}
}

# Any scope, inside an Ecumenical Council (needs scope:host and scope:activity). The council's host is
# an AI (not a player's theological puppet) and the debated proposal would change a protected rite:
# making a tenet core or changing a doctrine changes the mainline rite, and, with the game rule set to
# block it, prohibiting a tenet that is a core tenet of a protected rite.
prr_council_proposal_blocked_trigger = {
	scope:host = { prr_is_player_or_player_puppet_trigger = no }
	OR = {
		AND = {
			OR = {
				scope:activity.var:current_debate_proposal ?= flag:make_core
				scope:activity.var:current_debate_proposal ?= flag:doctrine_change
			}
			scope:host.faith.main_rite = { prr_rite_is_protected_trigger = yes }
		}
		AND = {
			has_game_rule = prr_prohibit_blocked
			scope:activity.var:current_debate_proposal ?= flag:prohibit
			scope:activity.var:current_topic ?= {
				prr_tenet_is_core_in_protected_rite_trigger = { FAITH = scope:host.faith }
			}
		}
	}
}
