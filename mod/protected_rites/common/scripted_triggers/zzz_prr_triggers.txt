# Character scope. An AI Catholic Head of Faith can proclaim Armed Pilgrimages a core tenet of the
# mainline rite (pam_proclaim_crusade_age_tenet_decision).
pam_ai_can_proclaim_armed_pilgrimages_core_trigger = {
	faith = {
		this = faith:catholic
		has_doctrine = doctrine_spiritual_head
		main_rite = {
			NOT = {
				has_tenet_status = {
					tenet = tenet_armed_pilgrimages
					status = core
				}
			}
		}
	}
	# PROTECTED RITES: the AI does not make a tenet core in a protected mainline rite
	faith.main_rite = { prr_rite_is_protected_trigger = no }
}

# Character scope. The Islamic counterpart: Struggle and Submission as a core tenet of the mainline rite.
pam_ai_can_proclaim_struggle_submission_core_trigger = {
	faith.religion = {
		this = religion:islam_religion
		exists = var:variable_ghw_unlocked # You've entered the Age of Crusades
	}
	faith = {
		main_rite = {
			NOT = {
				has_tenet_status = {
					tenet = tenet_struggle_submission
					status = core
				}
			}
		}
	}
	# PROTECTED RITES: the AI does not make a tenet core in a protected mainline rite
	faith.main_rite = { prr_rite_is_protected_trigger = no }
}

# Tenet scope. A Head of Faith (scope:actor) can prohibit the tenet with a bull or decision.
pam_tenet_can_be_prohibited_trigger = {
	save_temporary_scope_as = pam_tenet_candidate
	custom_tooltip = {
		text = pam_tenet_stance_can_be_prohibited_tt
		scope:actor.rite = {
			NOR = {
				has_tenet_status = { tenet = scope:pam_tenet_candidate status = prohibited }
				has_tenet_status = { tenet = scope:pam_tenet_candidate status = core }
			}
		}
	}
	# PROTECTED RITES: with the game rule set to block it, an AI Head of Faith cannot prohibit a core
	# tenet of a protected rite
	trigger_if = {
		limit = {
			has_game_rule = prr_prohibit_blocked
			scope:actor = { is_ai = yes }
		}
		NOT = { prr_tenet_is_core_in_protected_rite_trigger = { FAITH = scope:actor.faith } }
	}
}

# Tenet scope. As above, with the Head of Faith as root.
pam_tenet_can_be_prohibited_no_actor_trigger = {
	save_temporary_scope_as = pam_tenet_candidate
	custom_tooltip = {
		text = pam_tenet_stance_can_be_prohibited_tt
		root.rite = {
			NOR = {
				has_tenet_status = { tenet = scope:pam_tenet_candidate status = prohibited }
				has_tenet_status = { tenet = scope:pam_tenet_candidate status = core }
			}
		}
	}
	# PROTECTED RITES: with the game rule set to block it, an AI Head of Faith cannot prohibit a core
	# tenet of a protected rite
	trigger_if = {
		limit = {
			has_game_rule = prr_prohibit_blocked
			root = { is_ai = yes }
		}
		NOT = { prr_tenet_is_core_in_protected_rite_trigger = { FAITH = root.faith } }
	}
}
