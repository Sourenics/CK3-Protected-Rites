# Protected Rites - on_actions. Vanilla on_actions are only extended through on_actions lists,
# so no vanilla effect block is overridden.

on_rite_created = {
	on_actions = { prr_on_rite_created }
}

on_rite_updated = {
	on_actions = { prr_on_rite_updated }
}

on_game_start_after_lobby = {
	on_actions = { prr_mark_mod_loaded }
}

yearly_global_pulse = {
	on_actions = { prr_mark_mod_loaded }
}

# Root is the creator. Marks the new rite if a player, or a player's theological puppet, founded it.
prr_on_rite_created = {
	effect = {
		if = {
			limit = { prr_is_player_or_player_puppet_trigger = yes }
			rite ?= {
				if = {
					limit = { rite_founder ?= root }
					set_variable = prr_player_founded_rite
				}
			}
		}
	}
}

# Root is the rite. Tells the players of a protected rite when it changed without a player edit
# while its Head of Rite is AI, so a change the mod could not stop never goes unnoticed.
prr_on_rite_updated = {
	trigger = {
		exists = scope:changed_tenets_or_doctrines
		NOT = { exists = scope:consume_rite_edit_token }
		# Ecumenical councils change the mainline rite on purpose, with their own notification
		NOT = { faith = { has_variable = ecumenical_council_ongoing } }
		prr_rite_is_protected_trigger = yes
		head_of_rite ?= { is_ai = yes }
	}
	effect = {
		head_of_rite = { save_scope_as = prr_head }
		debug_log = "Protected Rites: a protected rite changed without a player edit while its Head of Rite is AI"
		every_player = {
			limit = { rite = root }
			save_scope_as = prr_player
			send_interface_message = {
				type = msg_rite_updated
				title = prr_msg_rite_changed_title
				desc = prr_msg_rite_changed_desc
				left_icon = scope:prr_head
			}
		}
	}
}

# Incompatible mods. Each loaded mod marks itself at game start and every year for a little less than a
# year, so a mark disappears soon after its mod is removed. A day later, if Head of Faith Council Veto
# (1.1.1 or later) has marked itself too, the players get an event once per game. Only this mod warns.
prr_mark_mod_loaded = {
	effect = {
		set_global_variable = {
			name = prr_mod_loaded
			value = yes
			days = 360
		}
		trigger_event = {
			on_action = prr_check_incompatible_mods
			days = 1
		}
	}
}

prr_check_incompatible_mods = {
	trigger = {
		has_global_variable = prr_mod_loaded
		has_global_variable = hfcv_mod_loaded
		NOT = { has_global_variable = prr_hfcv_incompatible_warned }
	}
	effect = {
		set_global_variable = {
			name = prr_hfcv_incompatible_warned
			value = yes
		}
		debug_log = "Protected Rites: Head of Faith Council Veto is also loaded; both mods override the Ecumenical Council vote tally"
		every_player = {
			trigger_event = prr_compat.0001
		}
	}
}
